  <div id="wrap">
    <h1>診断ログ</h1>
    <div id="log">ページ読み込み：OK（HTMLは表示できています）</div>
    <div class="hint">※ ここにステップごとの結果が表示されます。</div>
    <div id="stage">3D初期化前（ここに描画されます）</div>
  </div>

  <script>
    const logBox = document.getElementById('log');
    const stage = document.getElementById('stage');
    const log = (m)=>{ logBox.textContent += '\n' + m; console.log(m); };

    function loadScript(src){
      return new Promise((res, rej)=>{
        const s = document.createElement('script');
        s.src = src;
        s.onload = ()=>res(src);
        s.onerror = ()=>rej(new Error('load failed: '+src));
        document.head.appendChild(s);
      });
    }

    (async()=>{
      log('Three.js 読み込み開始（jsDelivr）…');
      try{
        await loadScript('https://cdn.jsdelivr.net/npm/three@0.152.2/build/three.min.js');
        log('OK: jsDelivr から three.min.js 読み込み成功');
      }catch(e){
        log('NG: jsDelivr 失敗 → 代替CDN(unpkg)を試します');
        await loadScript('https://unpkg.com/three@0.152.2/build/three.min.js');
        log('OK: unpkg から three.min.js 読み込み成功');
      }

      // OrbitControls（グローバル版）
      try{
        log('OrbitControls 読み込み開始（jsDelivr）…');
        await loadScript('https://cdn.jsdelivr.net/npm/three@0.152.2/examples/js/controls/OrbitControls.js');
        log('OK: jsDelivr から OrbitControls 読み込み成功');
      }catch(e){
        log('NG: jsDelivr 失敗 → 代替CDN(unpkg)を試します');
        await loadScript('https://unpkg.com/three@0.152.2/examples/js/controls/OrbitControls.js');
        log('OK: unpkg から OrbitControls 読み込み成功');
      }

      // ====== ここから3D初期化（最小）======
      try{
        log('3D初期化を開始します…');
        const scene = new THREE.Scene();
        scene.background = new THREE.Color(0x87ceeb);

        const camera = new THREE.PerspectiveCamera(75, stage.clientWidth / stage.clientHeight, 0.1, 1000);
        camera.position.set(0, 1.7, 5);

        const renderer = new THREE.WebGLRenderer({antialias:true});
        renderer.setSize(stage.clientWidth, stage.clientHeight);
        stage.innerHTML = '';
        stage.appendChild(renderer.domElement);

        const grid = new THREE.GridHelper(20, 20, 0x000000, 0x00ff00);
        scene.add(grid);

        const geo = new THREE.SphereGeometry(0.3, 32, 32);
        const mat = new THREE.MeshStandardMaterial({color:0xff0000});
        const ball = new THREE.Mesh(geo, mat);
        ball.position.set(0, 0.3, -3);
        scene.add(ball);

        const light = new THREE.DirectionalLight(0xffffff, 1);
        light.position.set(10,10,10);
        scene.add(light);

        const controls = new THREE.OrbitControls(camera, renderer.domElement);
        controls.target.set(0, 0.3, -3);
        controls.enablePan = false;

        const onResize = ()=>{
          const w = stage.clientWidth, h = stage.clientHeight;
          camera.aspect = w/h; camera.updateProjectionMatrix();
          renderer.setSize(w,h);
        };
        addEventListener('resize', onResize);

        function animate(){
          requestAnimationFrame(animate);
          controls.update();
          renderer.render(scene, camera);
        }
        animate();
        log('3D初期化：成功 🎉 画面下部にグリッドと赤い球が見えるはずです。');

      }catch(err){
        log('3D初期化でエラー：' + (err && err.message ? err.message : err));
        stage.textContent = '初期化エラー（ログを確認してください）';
      }
    })().catch(err=>{
      log('致命的エラー：'+ (err && err.message ? err.message : err));
    });
  </script>
